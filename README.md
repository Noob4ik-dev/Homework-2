# Homework-2
# Условие задачи
Человеку нужно съездить из Лондона в расположенный в 390 милях Эдинбург. Он может ехать на авто-мобиле "Роллс-Ройс" либо на автомобиле марки "Форд-Эскорт". "Роллс-Ройс" расходует 1 галлон бензина на каждые 15 миль пути. "Форд-Эскорт" расходует 1 галлон бензина на каждые 36 миль пути. Сколько будет стоить поездка в Эдинбург на "Роллс-Ройсе", если стоимость 1 галлона бензина составляет x фунтов? Сколько денег он сбережёт, если вместо этого поедет на машине марки "Форд-Эскорт".
# 1. Алгоритм и текстовое описание блок-схемы
Алгоритм

1.Начало

2. Объявить константы:

     DISTANCE = 390.0 (миль)— расстояние от Лондона до Эдинбурга.

     RR_MPG = 15.0 (миль / галлон) — расход Rolls-Royce.

     FORD_MPG = 36.0 (миль / галлон) — расход Ford-Escort.

3. Задать исходные данные:

    x — стоимость 1 галлона бензина (в фунтах стерлингов).

4. Рассчитать расход топлива:

    rr_gallons = DISTANCE / RR_MPG — расход Rolls-Royce в галлонах.

    ford_gallons = DISTANCE / FORD_MPG — расход Ford-Escort в галлонах.

5. Вычислить стоимость поездки:

   cost_rr = rr_gallons * x — стоимость поездки на Rolls-Royce.

   cost_ford = ford_gallons * x — стоимость поездки на Ford-Escort.

6. Вычислить экономию:

   savings = cost_rr - cost_ford — сбережённая сумма.

7. Вывести результаты расчётов с форматированным подставлением исходных данных и итогов в текст.

8. Конец.

# 2. Реализация программы.

```c

#include <stdio.h>
#include <locale.h>

#ifdef _WIN32
#include <windows.h>
#endif

int main(void) {
#ifdef _WIN32
    SetConsoleOutputCP(65001);
    SetConsoleCP(65001);
#endif
    setlocale(LC_ALL, "Russian");

    const double DISTANCE = 390.0;
    const double RR_MPG = 15.0;
    const double FORD_MPG = 36.0;

    double x = 1.50;

    double rr_gallons = DISTANCE / RR_MPG;
    double ford_gallons = DISTANCE / FORD_MPG;

    double cost_rr = rr_gallons * x;
    double cost_ford = ford_gallons * x;

    double savings = cost_rr - cost_ford;

    printf("Расчёт стоимости поездки\n");

    printf("Условие задачи и исходные данные:\n");
    printf("Человеку нужно съездить из Лондона в расположенный в %.0f милях Эдинбург.\n", DISTANCE);
    printf("Он может ехать на автомобиле Rolls-Royce (расход 1 галлон на %.0f миль) \n", RR_MPG);
    printf("либо на автомобиле Ford Escort (расход 1 галлон на %.0f миль).\n", FORD_MPG);
    printf("Стоимость 1 галлона бензина составляет x = %.2f фунтов.\n\n", x);

    printf("Результаты расчётов:\n");
    printf("1. Расход топлива Rolls-Royce: %.2f галлонов.\n", rr_gallons);
    printf("2. Расход топлива Ford Escort: %.2f галлонов.\n", ford_gallons);
    printf("3. Стоимость поездки на Rolls-Royce составит: %.2f фунтов.\n", cost_rr);
    printf("4. Стоимость поездки на Ford Escort составит: %.2f фунтов.\n", cost_ford);
    printf("5. Поехав на Ford Escort вместо Rolls-Royce, человек сбережет: %.2f фунтов.\n\n", savings);

    return 0;
}
```
